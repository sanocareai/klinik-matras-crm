# backend IDENTIK (kecuali DUA berkas tes pemeriksa kontrak metrik, dikecualikan eksplisit dari daftar hash — bukan kode runtime): kode di dalam container SETELAH switch == folder release baru == kode di container SEBELUM switch (image sama, backend/ tidak berubah)
docker exec "$CID_NEW" sh -c 'cd /app && { for x in src prisma scripts tests; do [ -d "$x" ] && find "$x" -type f; done; } | LC_ALL=C sort | grep -vxE "tests/(financeKontrakMetrik\.test\.js|helpers/pemeriksaMetrikUi\.js)" | xargs sha256sum && sha256sum package.json package-lock.json' > "$BK_DIR/container-sha-baru.txt" || die "gagal membaca kode di dalam container baru"
( cd "$NEW_DIR/backend" && { for x in src prisma scripts tests; do [ -d "$x" ] && find "$x" -type f; done; } | LC_ALL=C sort | grep -vxE "tests/(financeKontrakMetrik\.test\.js|helpers/pemeriksaMetrikUi\.js)" | xargs sha256sum && sha256sum package.json package-lock.json ) > "$BK_DIR/dir-sha-baru.txt"
grep -vE "  tests/(financeKontrakMetrik\.test\.js|helpers/pemeriksaMetrikUi\.js)$" "$BK_DIR/container-sha.txt" > "$BK_DIR/container-sha-tanpa-tes.txt"
cmp -s "$BK_DIR/container-sha-baru.txt" "$BK_DIR/dir-sha-baru.txt" || die "kode di container baru BERBEDA dari folder release baru"
cmp -s "$BK_DIR/container-sha-baru.txt" "$BK_DIR/container-sha-tanpa-tes.txt" || die "kode backend di container BERUBAH setelah switch (harus identik dengan sebelum switch)"
ok "backend identik: kode di container sebelum == sesudah switch == folder release baru, selain dua berkas tes pemeriksa kontrak metrik ($(wc -l < "$BK_DIR/container-sha-baru.txt" | tr -d ' ') berkas)"
